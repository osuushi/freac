cmake_minimum_required(VERSION 3.20)
project(freac_kernel LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCASCADE 7.9.3 EXACT CONFIG REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
find_path(BOOST_INCLUDE_DIR boost/property_tree/json_parser.hpp REQUIRED)
add_executable(freac-kernel main.cpp shell.cpp shell-validation.cpp shell-canonical.cpp extrude-draft.cpp cleanup.cpp delete-topology.cpp projection.cpp geometry.cpp revolve.cpp screw-sweep.cpp axial-sections.cpp presentation.cpp transform.cpp booleans.cpp edge-finishes.cpp face-offset.cpp offset-contacts.cpp face-chains.cpp offset-validation.cpp blend-recognition.cpp blend-resize.cpp)
target_include_directories(freac-kernel SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR} ${BOOST_INCLUDE_DIR})
target_sources(freac-kernel PRIVATE boundary-move.cpp boundary-selection.cpp boundary-edges.cpp boundary-faces.cpp boundary-cylinder.cpp)
target_sources(freac-kernel PRIVATE offset-collapse.cpp)
target_sources(freac-kernel PRIVATE offset-geometry.cpp offset-boundaries.cpp offset-result.cpp)
target_sources(freac-kernel PRIVATE measurement.cpp measurement-gaps.cpp measurement-planar.cpp)
target_sources(freac-kernel PRIVATE sketch-curve.cpp)
target_sources(freac-kernel PRIVATE boolean-periodic.cpp)
target_sources(freac-kernel PRIVATE extrude-twist.cpp extrude-twist-curves.cpp extrude-symmetric.cpp)
target_sources(freac-kernel PRIVATE plane-cut.cpp path-sweep.cpp)
target_link_libraries(freac-kernel PRIVATE TKFeat)
target_link_libraries(freac-kernel PRIVATE TKShHealing TKOffset TKFillet TKPrim TKBO TKBool TKMesh TKTopAlgo TKGeomAlgo TKGeomBase TKBRep TKG3d TKG2d TKMath TKernel)
set_target_properties(freac-kernel PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER ${config} config_upper)
  set_target_properties(freac-kernel PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config_upper} ${CMAKE_BINARY_DIR}/bin)
endforeach()
option(FREAC_KERNEL_TESTS "Build focused native kernel checks" OFF)
if(FREAC_KERNEL_TESTS)
  enable_testing()
  add_executable(sweep-distance-test tests/sweep-distance.cpp)
  target_include_directories(sweep-distance-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
  target_include_directories(sweep-distance-test SYSTEM PRIVATE ${OpenCASCADE_INCLUDE_DIR})
  target_link_libraries(sweep-distance-test PRIVATE TKPrim TKTopAlgo TKGeomAlgo TKGeomBase TKBRep TKG3d TKG2d TKMath TKernel)
  add_test(NAME sweep-distance COMMAND sweep-distance-test)
endif()
